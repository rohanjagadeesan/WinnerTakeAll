#!/bin/bash
#SBATCH --job-name=comp_nch_3mb
#SBATCH --account=commons              
#SBATCH --partition=commons            
#SBATCH --ntasks=1                
#SBATCH --cpus-per-task=1             
#SBATCH --mem-per-cpu=32G           
#SBATCH --time=04:00:00            
#SBATCH --mail-user=rj65@rice.edu 
#SBATCH --mail-type=ALL            

#SBATCH --array=0-17
#SBATCH --output=logs/comp/comp_nch_%A_%a.out       
#SBATCH --error=logs/comp/comp_nch_%A_%a.err 

# Move your execution context safely into scratch
cd $SHARED_SCRATCH/$USER/wta_scratch

# Call your safe, permanent home directory virtual environment
source $HOME/.venv/bin/activate

echo "comparing clustertypes in 3mb nch60_pt500"
echo "Starting Task ID $SLURM_ARRAY_TASK_ID at : $(date)"
echo "Running on hosts: $SLURM_NODELIST"
echo "Running on $SLURM_NNODES nodes."
echo "Running on $SLURM_NPROCS processors."
echo "Current WORKDIR is : $SLURM_SUBMIT_DIR"

if [[ -d $SHARED_SCRATCH/$USER && -w $SHARED_SCRATCH/$USER ]]; then
    
    # target ALL BATCHES
    INPUT_FILE="$SHARED_SCRATCH/$USER/wta_scratch/data/3mb/nch60_pt500/"
    OUTPUT_DIR="$SHARED_SCRATCH/$USER/wta_scratch/intermediate_output/3mb_nch60/nch_comparison_premerge/"

    # target specific batch
    BATCH_NUM=$SLURM_ARRAY_TASK_ID


    python -u "$SHARED_SCRATCH/$USER/wta_scratch/clustertype_comparisons.py" \
    --input "$INPUT_FILE" \
    --outdir "$OUTPUT_DIR" \
    --batchnum $BATCH_NUM
fi
