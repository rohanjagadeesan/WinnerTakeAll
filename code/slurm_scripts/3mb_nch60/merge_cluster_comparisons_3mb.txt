# Move your execution context safely into scratch
cd $SHARED_SCRATCH/$USER/wta_scratch

# Call your safe, permanent home directory virtual environment
source $HOME/.venv/bin/activate

echo "Merging 3mb cluster type comparisons"
echo "Starting at : `date`"
echo "Running on hosts: $SLURM_NODELIST"
echo "Running on $SLURM_NNODES nodes."

if [[ -d $SHARED_SCRATCH/$USER && -w $SHARED_SCRATCH/$USER ]]; then
    cd $SHARED_SCRATCH/$USER/wta_scratch
    source $HOME/.venv/bin/activate

    # target merged signals
    INPUT_FILE="$SHARED_SCRATCH/$USER/wta_scratch/intermediate_output/3mb_nch60/nch_comparison_premerge/"
    OUTPUT_DIR="$SHARED_SCRATCH/$USER/wta_scratch/output/3mb_nch60/"

    # submit sbatch from the same directory as the .py file
    python -u "$SHARED_SCRATCH/$USER/wta_scratch/merge_cluster_comps.py" --input "$INPUT_FILE" --outdir "$OUTPUT_DIR"
fi
