#!/bin/bash
#SBATCH --job-name=WTA_all_batches_0mb_inc
#SBATCH --account=commons              
#SBATCH --partition=commons            
#SBATCH --ntasks=1                
#SBATCH --cpus-per-task=1             
#SBATCH --mem-per-cpu=16G           
#SBATCH --time=08:00:00            
#SBATCH --mail-user=rj65@rice.edu 
#SBATCH --mail-type=ALL            

#SBATCH --output=logs/per_batch_signal/sig_%j.out       
#SBATCH --error=logs/per_batch_signal/sig_%j.err 



cd $SHARED_SCRATCH/$USER/wta_scratch # Move your execution context safely into scratch
source $HOME/.venv/bin/activate # activate venv

echo "analysing ALL BATCHES in 0mb inclusive"
echo "Starting Task ID $SLURM_ARRAY_TASK_ID at : $(date)"
echo "Running on hosts: $SLURM_NODELIST"
echo "Running on $SLURM_NNODES nodes."
echo "Running on $SLURM_NPROCS processors."
echo "Current WORKDIR is : $SLURM_SUBMIT_DIR"

if [[ -d $SHARED_SCRATCH/$USER && -w $SHARED_SCRATCH/$USER ]]; then

    # target ALL BATCHES
    INPUT_FILE="$SHARED_SCRATCH/$USER/wta_scratch/data/0mb/inclusive/"
    OUTPUT_DIR="$SHARED_SCRATCH/$USER/wta_scratch/intermediate_output/new_radius/0mb_inclusive/premerge/"

    # target specific batch - only 0 for inclusive

    # submit sbatch from the same directory as the .py file
    python -u "$SHARED_SCRATCH/$USER/wta_scratch/all_batch_2pc_maker.py" \
    --input "$INPUT_FILE" \
    --outdir "$OUTPUT_DIR" \
    --batchnum 0
fi
